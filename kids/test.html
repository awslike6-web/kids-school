async function askGeminiNextGen(childText, apiKey) { document.getElementById('fairy-reply').innerText = "다람냥 생각 중... 💭"; // 🌟 [2026 정석 주소] 1.5는 폐기되었습니다. 최신 2.0/2.5 플래시 정식 유료 게이트웨이 주소입니다. // 만약 계정 세팅에 따라 2.5가 안 붙으면 주소의 'gemini-2.5-flash'를 'gemini-2.0-flash'로만 바꿔주세요. const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; try { const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ // 🌟 최신 2.x 세대는 systemInstruction 스펠링과 구조가 엄격해졌습니다. systemInstruction: { parts: [{ text: FAIRY_PROMPT }] }, contents: [{ role: "user", parts: [{ text: childText }] }], // 🌟 온도를 0.7로 안정화하고 대사 길이를 제한해 버벅임을 최소화합니다. generationConfig: { temperature: 0.7, maxOutputTokens: 100 } }) }); // 서버 응답 상태 체크 if (!response.ok) { const errorText = await response.text(); console.error("구글 서버 에러 메시지:", errorText); throw new Error(`서버가 문을 닫음 (코드: ${response.status})`); } const data = await response.json(); // 최신 2.x 응답 데이터 파싱 안전장치 if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].text) { const fairyText = data.candidates[0].content.parts[0].text; document.getElementById('fairy-reply').innerText = fairyText; // 다람냥 목소리 출력! speakResponse(fairyText); } else { throw new Error("구글 응답 구조가 바뀌었거나 파싱에 실패했습니다."); } } catch (error) { console.error("Gemini API 호출 에러:", error); document.getElementById('fairy-reply').innerText = `⚠️ 연결 실패! (${error.message})`; resetButton(); } }
